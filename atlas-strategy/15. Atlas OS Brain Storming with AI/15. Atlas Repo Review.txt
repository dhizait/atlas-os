def create_file(path, content):
    with open(path, "w", encoding="utf-8") as f:
        f.write(content)

def main():
    print("Scaffolding ATLAS OS...")
    for d in DIRS:
        os.makedirs(os.path.join(BASE, d), exist_ok=True)
        # add __init__.py for python packages
        if "drivers" in d or "platform" in d:
            create_file(os.path.join(BASE, d, "__init__.py"), "")

    for file, content in FILES.items():
        create_file(os.path.join(BASE, file), content)

    print(f"\nDONE. Repo created at./{BASE}/")
    print("Next steps:")
    print("1. cd atlas-os")
    print("2. docker-compose up -d")
    print("3. python scripts/run_driver_batch.py core.interest_accrual")

if __name__ == "__main__":
    main()
