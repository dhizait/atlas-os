def sanitize_filename(name):

def docx_to_text(docx_path):
    try:
        doc = Document(docx_path)
        lines = []
        for p in doc.paragraphs:
            t = p.text.strip()
            if t:
                lines.append(t)
        for tbl in doc.tables:
            lines.append(f"[TABLE {len(tbl.rows)}x{len(tbl.columns)}]")
        return "\n\n".join(lines)
    except Exception as e:
        return f"[ERROR READING DOCX: {e}]"

def create_light_docx(source_path, target_path):
    shutil.copy(source_path, target_path)
    tmp_path = target_path + ".tmp"
    with zipfile.ZipFile(target_path, 'r') as zin:
        with zipfile.ZipFile(tmp_path, 'w', zipfile.ZIP_DEFLATED) as zout:
            for item in zin.namelist():
                if not item.startswith('word/media/'):
                    zout.writestr(item, zin.read(item))
    os.replace(tmp_path, target_path)

def backup_and_clear_extracts(cwd, log):
    extract_dir = os.path.join(cwd, "doc_extracts")
    backup_root = os.path.join(cwd, "doc_extracts_backup")
    if not os.path.exists(extract_dir):
        log.append("No existing doc_extracts found. Skipping backup.")
        os.makedirs(extract_dir, exist_ok=True)
        return
    if not os.listdir(extract_dir):
        log.append("Existing doc_extracts is empty. Skipping backup.")
        return
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_dir = os.path.join(backup_root, timestamp)
    os.makedirs(backup_root, exist_ok=True)
    try:
        shutil.move(extract_dir, backup_dir)
        log.append(f"BACKUP CREATED: Moved existing doc_extracts to doc_extracts_backup\\{timestamp}")
    except PermissionError:
        log.append("WARN: Could not backup - files in use. DELETING doc_extracts instead")
        shutil.rmtree(extract_dir, ignore_errors=True)
    os.makedirs(extract_dir, exist_ok=True)

def prune_old_backups(cwd, log, keep_n):
    if keep_n <= 0:
        log.append("Backup pruning disabled: keep_last_n_backups=0")
        return
    backup_root = os.path.join(cwd, "doc_extracts_backup")
    if not os.path.exists(backup_root): return
    backups = [d for d in os.listdir(backup_root) if os.path.isdir(os.path.join(backup_root, d))]
    backups.sort()
    to_delete = backups[:-keep_n] if len(backups) > keep_n else []
    for old in to_delete:
        shutil.rmtree(os.path.join(backup_root, old))
        log.append(f"PRUNED OLD BACKUP: {old}")
    if to_delete:
        log.append(f"Kept last {keep_n} backups. Pruned {len(to_delete)}")
    else:
        log.append(f"Backup count {len(backups)}. Within keep limit of {keep_n}")

"""
SYNOPSIS: NAVIGATION PANE EXTRACTOR v1.4.0 - BLOCK COUNTING SOURCE OF TRUTH
Purpose: Count ALL blocks p+tbl+sectPr. CSV columns: START_BLOCK/END_BLOCK
Fixes: PATH hierarchy, DEDUPE, EMPTY PARENTS, TABLE counting
"""
import zipfile
import xml.etree.ElementTree as ET
import sys, re, csv, json

NS = {'w': 'http://schemas.openxmlformats.org/wordprocessingml/2006/main'}
QN = lambda tag: f"{{{NS['w']}}}{tag}"

def get_text_from_para(p):
    texts = p.findall('.//w:t', NS)
    return ''.join([t.text for t in texts if t.text]).strip()

def get_outline_level(p):
    pPr = p.find('w:pPr', NS)
    if pPr is None: return None
    outline = pPr.find('w:outlineLvl', NS)
    if outline is not None:
        val = int(outline.get(QN('val')))
        return val + 1
    return None

def extract_nav_pane(docx_path):
    nav_raw = []
    counters = [0] * 10
    block_index = 0
    path_stack = ["ROOT"]

    with zipfile.ZipFile(docx_path) as z:
        xml_content = z.read('word/document.xml')

    root = ET.fromstring(xml_content)
    body = root.find('w:body', NS)
    all_blocks = list(body)
    total_blocks = len(all_blocks)

    for block in all_blocks:
        level = None
        text = ""

        if block.tag == QN('p'):
            level = get_outline_level(block)
            text = get_text_from_para(block)

        if level is None or not text:
            block_index += 1
            continue

        text = re.sub(r'\s+\d+$', '', text)
        counters[level] += 1
        for j in range(level + 1, 10): counters[j] = 0
        number_parts = [str(counters[k]) for k in range(1, level + 1) if counters[k] > 0]
        unique_number = ".".join(number_parts)
        parent_number = ".".join(number_parts[:-1]) if len(number_parts) > 1 else ""

        # FIX: PATH HIERARCHY
        path_stack = path_stack[:level-1]
        path_stack.append(text)
        full_path = " > ".join(path_stack[1:])

        nav_raw.append({
            "heading_block": block_index,
            "level": level,
            "number": unique_number,
            "parent_number": parent_number,
            "title": text,
            "path": full_path
        })
        block_index += 1

    # FIX: DEDUPE. Keep first occurrence of each number
    nav = []
    seen_numbers = set()
    for item in nav_raw:
        if item['number'] not in seen_numbers:
            nav.append(item)
            seen_numbers.add(item['number'])

    # FIX: Use BLOCKS for start/end. Empty parents now get 1 block
    for i, item in enumerate(nav):
        item['index'] = i + 1
        item['start_block'] = item['heading_block']
        if i + 1 < len(nav):
            item['end_block'] = nav[i+1]['heading_block']
        else:
            item['end_block'] = total_blocks

    out_csv = docx_path.replace(".docx", "_NAV_PANE.csv")
    with open(out_csv, "w", encoding="utf-8", newline='') as f:
        writer = csv.writer(f)
        writer.writerow(["INDEX", "NUMBER", "PARENT_NUMBER", "LEVEL", "HEADING_BLOCK", "START_BLOCK", "END_BLOCK", "TITLE", "PATH"])
        for item in nav:
            writer.writerow([item['index'], item['number'], item['parent_number'], item['level'], item['heading_block'], item['start_block'], item['end_block'], item['title'], item['path']])

    out_json = docx_path.replace(".docx", "_NAV_INDEX.json")
    with open(out_json, "w", encoding="utf-8") as f:
        json.dump(nav, f, indent=2, ensure_ascii=False)

    print(f"[NAV-MODE] CSV: {out_csv}")
    print(f"[NAV-MODE] JSON: {out_json}")
    print(f"[NAV-MODE] Found {len(nav)} headings. Total blocks in doc: {total_blocks}")
    return nav, total_blocks

if __name__ == "__main__":
    docx_path = sys.argv[1]
    extract_nav_pane(docx_path)

"""
SYNOPSIS: DOCX SLICE WRITER v1.6.4
Purpose: Extract by cloning body XML using BLOCK indexes from NAV
Fixes: CT_Body bug + 1-block sections + BLOCK counting to match NAV
"""
import os, zipfile, shutil, tempfile
from docx import Document
from docx.oxml.ns import qn
from copy import deepcopy

QN = qn # alias

def save_slice_to_docx(master_docx_path, toc_node, output_dir, filename=None):
    # READ FROM NAV: use blocks now
    start_b = toc_node['start_block']
    end_b = toc_node['end_block']

    if end_b < start_b:
        print(f"WARN: Skipping '{toc_node['title']}' - invalid range {start_b} to {end_b}")
        return None
    if end_b == start_b:
        end_b = start_b + 1

    safe_title = toc_node['title'].replace("/","-").replace(":","-").replace("\\","-")[:40]
    filename = f"{toc_node['number']}_{safe_title}.docx"
    filename = filename.replace("/", "-").replace(":", "-").replace("\\", "-")
    out_path = os.path.join(output_dir, filename)

    tmp_dir = tempfile.mkdtemp()
    tmp_docx = os.path.join(tmp_dir, "temp.docx")
    shutil.copy(master_docx_path, tmp_docx)

    with zipfile.ZipFile(tmp_docx, 'r') as zin:
        with zipfile.ZipFile(out_path, 'w') as zout:
            for item in zin.namelist():
                if item!= 'word/document.xml':
                    zout.writestr(item, zin.read(item))
                else:
                    doc = Document(tmp_docx)
                    new_doc = Document()
                    new_body = new_doc.element.body
                    new_body.clear()

                    # Add PATH header
                    p = new_doc.add_paragraph()
                    p.add_run(f"PATH: {toc_node['path']}").bold = True

                    block_idx = 0
                    for element in doc.element.body:
                        # Count ALL blocks: p, tbl, sectPr
                        if element.tag in [QN('w:p'), QN('w:tbl'), QN('w:sectPr')]:
                            if start_b <= block_idx < end_b:
                                new_body.append(deepcopy(element))
                            block_idx += 1

                    zout.writestr(item, new_doc.element.xml.encode())

    shutil.rmtree(tmp_dir)
    return out_path

"""
SYNOPSIS: DOCX TOC SLICER v1.5.18
Purpose: Parses DOCX and maps each TOC entry to paragraph start/end ranges using bookmarks
         v1.5.18 FIX: Handles duplicate TOC titles/bookmarks by tracking occurrence count
"""
import re
from docx import Document
from docx.oxml.ns import qn
from collections import defaultdict

def get_toc_slices(docx_path):
    doc = Document(docx_path)
    toc_entries = []
    bookmark_counts = defaultdict(int) # v1.5.18 FIX: track duplicates

    # Pass 1: Get TOC entries and assign them a unique occurrence ID
    for i, p in enumerate(doc.paragraphs):
        for run in p._p.iter():
            if run.tag == qn('w:hyperlink'):
                anchor = run.get(qn('w:anchor'))
                if anchor and anchor.startswith('_Toc'):
                    bookmark_counts[anchor] += 1
                    unique_anchor = f"{anchor}__{bookmark_counts[anchor]}" # e.g _Toc123__1, _Toc123__2

                    # Clean title: remove page numbers and tabs
                    title = re.sub(r'\s+\d+$', '', p.text).strip()
                    title = re.sub(r'\t+', ' ', title).strip()
                    if title:
                        toc_entries.append({
                            "toc_para_index": i,
                            "title": title,
                            "bookmark": anchor,
                            "unique_bookmark": unique_anchor,
                            "occurrence": bookmark_counts[anchor]
                        })
                    break

    # Pass 2: Build map of ALL bookmarkStart occurrences in the document
    bookmark_to_paras = defaultdict(list)
    for j, p in enumerate(doc.paragraphs):
        for b in p._p.iter():
            if b.tag == qn('w:bookmarkStart'):
                bname = b.get(qn('w:name'))
                if bname and bname.startswith('_Toc'):
                    bookmark_to_paras[bname].append(j)

    print(f"DEBUG TOC: Found {len(toc_entries)} TOC entries")
    dom = []
    for i, entry in enumerate(toc_entries):
        anchor = entry['bookmark']
        occ = entry['occurrence'] - 1 # 0-based index

        # Get the correct occurrence of the start bookmark
        para_list = bookmark_to_paras.get(anchor, [entry['toc_para_index']])
        start_para = para_list[occ] if occ < len(para_list) else para_list[-1]

        # Get the correct occurrence of the END bookmark = next TOC entry
        if i + 1 < len(toc_entries):
            next_anchor = toc_entries[i+1]['bookmark']
            next_occ = toc_entries[i+1]['occurrence'] - 1
            next_para_list = bookmark_to_paras.get(next_anchor, [len(doc.paragraphs)])
            end_para = next_para_list[next_occ] if next_occ < len(next_para_list) else next_para_list[-1]
        else:
            end_para = len(doc.paragraphs)

        print(f"DEBUG: [{i}] '{entry['title'][:40]}' start={start_para} end={end_para} len={end_para-start_para} bm={anchor}#{entry['occurrence']}")
        dom.append({
            "toc_idx": i,
            "node_name": entry['title'],
            "start_para": start_para,
            "end_para": end_para
        })
    return dom

"""
SYNOPSIS: TREE JOINER v1.5.12
Purpose: Join tree.txt nodes to TOC entries with 3 state marking: EXTRACTED, COVERED, ORPHAN
         Handles UTF-16 and UTF-8 input
Deps: re
"""
import re

def parse_tree_file(tree_path, encoding='utf-8'):
    """Parse tree.txt to list of nodes with id, name, level, parent_id, original_line"""
    nodes = []
    stack = [] # stack of node_ids by level
    with open(tree_path, 'r', encoding=encoding, errors='ignore') as f:
        for line in f:
            raw_line = line.rstrip('\n')
            if '---' not in raw_line or '.bat' in raw_line:
                continue

            # Level from indentation. 4 spaces = 1 level
            level = (len(raw_line) - len(raw_line.lstrip())) // 4 + 1

            match = re.search(r'---\s*([\d\.]+)\s*(.*)', raw_line)
            if not match:
                continue
            node_id = match.group(1).strip()
            name = match.group(2).strip()

            # Maintain parent stack
            while len(stack) >= level:
                stack.pop()
            parent_id = stack[-1] if stack else None
            stack.append(node_id)

            nodes.append({
                "id": node_id,
                "node_name": name,
                "level": level,
                "parent_id": parent_id,
                "toc_idx": len(nodes), # TOC order assumed to match tree order
                "original_line": raw_line
            })
    return nodes

def find_parent_with_toc(node, toc_map, node_map):
    """Walk up parent_id until we find a node that has a TOC entry"""
    current = node
    while current:
        if current['toc_idx'] in toc_map:
            return current
        if not current['parent_id']:
            return None
        current = node_map.get(current['parent_id'])
    return None

def join_tree_to_toc(tree_nodes, toc_map):
    """
    Core logic for 3-state marking
    """
    node_map = {n['id']: n for n in tree_nodes}
    join_plan = {} # toc_idx -> node
    extracted_ids = []
    covered_ids = []
    orphan_ids = []

    # Pass 1: Identify all EXTRACTED nodes. These have direct TOC match
    for node in tree_nodes:
        if node['toc_idx'] in toc_map:
            join_plan[node['toc_idx']] = node
            extracted_ids.append(node['id'])

    extracted_toc_idxs = set(join_plan.keys())

    # Pass 2: For nodes without direct TOC, check if parent was extracted
    for node in tree_nodes:
        if node['id'] in extracted_ids:
            continue # already handled

        parent = find_parent_with_toc(node, toc_map, node_map)
        if parent and parent['toc_idx'] in extracted_toc_idxs:
            # Scenario 1: Covered by parent
            covered_ids.append(node['id'])
        else:
            # Scenario 2: True orphan
            orphan_ids.append(node['id'])

    # Pass 3: Build marked tree for topic_tree.log
    marked_tree_lines = []
    for node in tree_nodes:
        line = node['original_line']
        if node['id'] in extracted_ids:
            line = re.sub(r'---\s*' + re.escape(node['id']), f'--- [EXTRACTED] {node["id"]}', line, count=1)
        elif node['id'] in covered_ids:
            line = re.sub(r'---\s*' + re.escape(node['id']), f'--- [COVERED] {node["id"]}', line, count=1)
        elif node['id'] in orphan_ids:
            line = re.sub(r'---\s*' + re.escape(node['id']), f'--- [ORPHAN] {node["id"]}', line, count=1)
        marked_tree_lines.append(line)

    return join_plan, extracted_ids, covered_ids, orphan_ids, marked_tree_lines

###  R17AMR Wiki DOCX TOC Extractor - v1.5.6

This toolkit extracts sections from any structured R17AMR Wiki DOCX based on its TOC and Tree.txt.

Each TOC section becomes a standalone `.docx` named `id Title.docx` and is dumped flat into `./doc_extracts/` for direct 1-to-1 mapping to your Wiki tree. 
Before each run, the previous `doc_extracts` is time-stamped and backed up to `./doc_extracts_backup/`. Includes 1 master `_index.json` for AI/RAG ingestion.

#### Core Principle: Code Central, Config Local, Safe Runs
1.  **Code Central**: All Python libraries, `docx_driver.py`, `run.bat`, `bulk_run.bat` live once in `C:\R17AMR\docx-lib\`
2.  **Config Fallback**: `granularity.json` is optional per topic. Falls back to central `docx-lib\granularity.json`
3.  **Data Local**: `.docx`, `.tree.txt` live per topic folder
4.  **Output**: All extracts go to `./doc_extracts/`. Old extracts are auto-backed up.
5.  **Safety**: Auto-prune keeps only the last N backups to prevent disk bloat.

Location: `C:\R17AMR\`

Run from any topic directory. Requires: `python-docx`, `docxcompose`, `pytest`, `pywin32`
